module serializer (
    input        clk,
    input        rst_n,
    input        load_en,
    input        shift_en,
    input  [7:0] tx_data,
    input        par_bit,
    output       tx_out
);

    reg [10:0] sreg;

    assign tx_out = sreg[0];

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n)
            sreg <= 11'h7ff;
        else if (load_en)
            sreg <= {1'b1, par_bit, tx_data, 1'b0};
        else if (shift_en)
            sreg <= {1'b1, sreg[10:1]};
    end

endmodule
