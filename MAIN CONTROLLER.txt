module uart_tx_fsm (
    input        clk,
    input        rst_n,
    input        fifo_empty,
    input  [2:0] par_cfg,
    output reg   fifo_rd,
    output reg   load_str,
    output reg   shift_en,
    output reg [1:0] mux_ctrl,
    output       busy
);

    parameter IDLE  = 3'd0,
              LOAD  = 3'd1,
              START = 3'd2,
              DATA  = 3'd3,
              PAR   = 3'd4,
              STOP  = 3'd5;

    reg [2:0] state, next_state;
    reg [2:0] bit_cnt;

    wire has_parity = (par_cfg != 3'd0);

    assign busy = (state != IDLE);

    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state <= IDLE;
            bit_cnt <= 3'd0;
        end else begin
            state <= next_state;
            if (state == DATA)
                bit_cnt <= bit_cnt + 1'b1;
            else
                bit_cnt <= 3'd0;
        end
    end

    always @(*) begin
        next_state = state;
        fifo_rd  = 1'b0;
        load_str = 1'b0;
        shift_en = 1'b0;
        mux_ctrl = 2'b00;

        case (state)
            IDLE: begin
                if (!fifo_empty) begin
                    fifo_rd = 1'b1;
                    next_state = LOAD;
                end
            end

            LOAD: begin
                load_str = 1'b1;
                mux_ctrl = 2'b01;
                next_state = START;
            end

            START: begin
                shift_en = 1'b1;
                mux_ctrl = 2'b01;   // Fixed syntax error
                next_state = DATA;
            end

            DATA: begin
                shift_en = 1'b1;
                mux_ctrl = 2'b01;
                if (bit_cnt == 3'd7)
                    next_state = has_parity ? PAR : STOP;
            end

            PAR: begin
                shift_en = 1'b1;
                mux_ctrl = 2'b01;
                next_state = STOP;
            end

            STOP: begin
                mux_ctrl = 2'b00;
                next_state = IDLE;
            end

            default: next_state = IDLE;
        endcase
    end

endmodule
