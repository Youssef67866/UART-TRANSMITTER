module uart_tx_top (
    input        baud_clk,
    input        rst_n,
    input  [7:0] din,
    input        empty,
    input  [2:0] parity_mode,
    output       pop_fifo,
    output       tx_busy,
    output       txd
);

    wire calc_parity;
    wire shift_serial_out;
    wire [1:0] mux_sel;
    wire load_pulse;
    wire shift_pulse;

    parity_gen u_parity (
        .din      (din),
        .par_type (parity_mode),
        .par_out  (calc_parity)
    );

    uart_tx_fsm u_ctrl (
        .clk        (baud_clk),
        .rst_n      (rst_n),
        .fifo_empty (empty),
        .par_cfg    (parity_mode),
        .fifo_rd    (pop_fifo),
        .load_str   (load_pulse),
        .shift_en   (shift_pulse),
        .mux_ctrl   (mux_sel),
        .busy       (tx_busy)
    );

    serializer u_shift_reg (
        .clk        (baud_clk),
        .rst_n      (rst_n),
        .load_en    (load_pulse),
        .shift_en   (shift_pulse),
        .tx_data    (din),
        .par_bit    (calc_parity),
        .tx_out     (shift_serial_out)
    );

    output_mux u_mux (
        .s_data   (shift_serial_out),
        .mode_sel (mux_sel),
        .tx_line  (txd)
    );

endmodule
