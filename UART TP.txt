`timescale 1ns / 1ps

module uart_tx_top_tb;

    reg         baud_clk;
    reg         rst_n;
    reg  [7:0]  din;
    reg         empty;
    reg  [2:0]  parity_mode;
    wire        pop_fifo;
    wire        tx_busy;
    wire        txd;

    uart_tx_top uut (
        .baud_clk    (baud_clk),
        .rst_n       (rst_n),
        .din         (din),
        .empty       (empty),
        .parity_mode (parity_mode),
        .pop_fifo    (pop_fifo),
        .tx_busy     (tx_busy),
        .txd         (txd)
    );

    always #500 baud_clk = ~baud_clk;

    task send_byte(input [7:0] data, input [2:0] parity);
    begin
        $display("--- Sending byte 0x%h with parity mode %d ---", data, parity);
        din = data;
        parity_mode = parity;
        empty = 0;

        @(posedge pop_fifo);
        $display("  Data read by transmitter at time %t", $time);
        @(negedge pop_fifo);
        empty = 1;

        @(negedge tx_busy);
        $display("  Transmission of byte 0x%h complete at time %t", data, $time);
    end
    endtask

    initial begin
        $dumpfile("uart_tx.vcd");
        $dumpvars(0, uart_tx_top_tb);

        baud_clk = 0;
        rst_n    = 0;
        din      = 8'h00;
        empty    = 1;
        parity_mode = 3'd0;

        #1000;
        rst_n = 1;
        $display("Reset released at time %t", $time);

        #2000;

        send_byte(8'h55, 3'd0);
        send_byte(8'hAA, 3'd1);
        send_byte(8'h01, 3'd2);
        send_byte(8'hFF, 3'd3);

        #10000;
        $display("Simulation finished at time %t", $time);
        $finish;
    end

    always @(posedge baud_clk) begin
        $display("Time %t | txd=%b | busy=%b | pop=%b | state=%d | bit_cnt=%d | din=0x%h | parity=%d",
                 $time,
                 txd,
                 tx_busy,
                 pop_fifo,
                 uut.u_ctrl.state,
                 uut.u_ctrl.bit_cnt,
                 din,
                 parity_mode);
    end

    always @(posedge pop_fifo) begin
        $display("  -> pop_fifo asserted at time %t", $time);
    end
    always @(negedge tx_busy) begin
        $display("  -> tx_busy deasserted at time %t", $time);
    end

endmodule
