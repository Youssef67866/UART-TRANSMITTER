`timescale 1ns / 1ps

module uart_tx_top_tb;

    // ---------- DUT signals ----------
    reg         baud_clk;
    reg         rst_n;
    reg  [7:0]  din;
    reg         empty;
    reg  [2:0]  parity_mode;
    wire        pop_fifo;
    wire        tx_busy;
    wire        txd;

    // Instantiate the top module
    uart_tx_top uut (
        .baud_clk    (baud_clk),
        .rst_n       (rst_n),
        .din         (din),
        .empty       (empty),
        .parity_mode (parity_mode),
        .pop_fifo    (pop_fifo),
        .tx_busy     (tx_busy),
        .txd         (txd)
    );

    // ---------- Clock generation ----------
    // 1 MHz baud clock (period = 1000 ns)
    always #500 baud_clk = ~baud_clk;

    // ---------- Tasks ----------
    // Task to send one byte with given parity
    task send_byte(input [7:0] data, input [2:0] parity);
    begin
        $display("--- Sending byte 0x%h with parity mode %d ---", data, parity);
        // Load data and parity mode
        din = data;
        parity_mode = parity;
        empty = 0;                     // indicate data available

        // Wait until the FSM reads the data (pop_fifo asserted)
        @(posedge pop_fifo);
        $display("  Data read by transmitter at time %t", $time);
        // After the read cycle, deassert empty (FIFO now empty)
        @(negedge pop_fifo);
        empty = 1;

        // Wait until the transmitter finishes the entire frame
        @(negedge tx_busy);
        $display("  Transmission of byte 0x%h complete at time %t", data, $time);
    end
    endtask

    // ---------- Main stimulus ----------
    initial begin
        // Dump waveform for viewing
        $dumpfile("uart_tx.vcd");
        $dumpvars(0, uart_tx_top_tb);

        // Initialize inputs
        baud_clk = 0;
        rst_n    = 0;
        din      = 8'h00;
        empty    = 1;
        parity_mode = 3'd0;

        // Release reset after a few clock cycles
        #1000;
        rst_n = 1;
        $display("Reset released at time %t", $time);

        // Wait for initial idle state
        #2000;

        // Send test bytes with different parity modes
        send_byte(8'h55, 3'd0);   // no parity
        send_byte(8'hAA, 3'd1);   // odd
        send_byte(8'h01, 3'd2);   // even
        send_byte(8'hFF, 3'd3);   // mark

        // Wait a bit then finish
        #10000;
        $display("Simulation finished at time %t", $time);
        $finish;
    end

    // ---------- Monitoring ----------
    // Print all important signals on each rising clock edge
    always @(posedge baud_clk) begin
        $display("Time %t | txd=%b | busy=%b | pop=%b | state=%d | bit_cnt=%d | din=0x%h | parity=%d",
                 $time,
                 txd,
                 tx_busy,
                 pop_fifo,
                 uut.u_ctrl.state,
                 uut.u_ctrl.bit_cnt,
                 din,
                 parity_mode);
    end

    // Also monitor pop_fifo and tx_busy changes for clarity
    always @(posedge pop_fifo) begin
        $display("  -> pop_fifo asserted at time %t", $time);
    end
    always @(negedge tx_busy) begin
        $display("  -> tx_busy deasserted at time %t", $time);
    end

endmodule
