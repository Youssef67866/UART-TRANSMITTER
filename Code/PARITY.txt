module parity_gen (
    input  [7:0] din,
    input  [2:0] par_type,
    output reg   par_out
);

    parameter PAR_NONE  = 3'd0;
    parameter PAR_ODD   = 3'd1;
    parameter PAR_EVEN  = 3'd2;
    parameter PAR_MARK  = 3'd3;
    parameter PAR_SPACE = 3'd4;

    always @(*) begin
        case (par_type)
            PAR_ODD:   par_out = ~(^din);
            PAR_EVEN:  par_out = ^din;
            PAR_MARK:  par_out = 1'b1;
            PAR_SPACE: par_out = 1'b0;
            default:   par_out = 1'b1;
        endcase
    end

endmodule
